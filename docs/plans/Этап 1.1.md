## 1.1. Таблица mtr_items — справочник деталей (гибкая схема)

```sql
CREATE TABLE mtr_items (
    -- Уникальный идентификатор
    id BIGSERIAL PRIMARY KEY,
    
    -- Корпоративные коды
    mtr_code TEXT UNIQUE NOT NULL,
    ksm_code TEXT UNIQUE,
    card_id TEXT,
    
    -- Базовые поля (есть у всех)
    item_type TEXT NOT NULL,          -- тип: труба, задвижка, отвод, ...
    subtype TEXT,                     -- подтип: бесшовная, клиновая, ...
    name TEXT NOT NULL,               -- полное наименование
    designation TEXT,                 -- условное обозначение
    
    -- Универсальное хранилище для всех параметров (гибкость!)
    attributes JSONB NOT NULL DEFAULT '{}',
    
    -- Метаданные
    gost_tu TEXT,                     -- ГОСТ/ТУ
    standard TEXT,                    -- стандарт
    stock_qty NUMERIC DEFAULT 0,
    unit TEXT DEFAULT 'pcs',
    is_synthetic BOOLEAN DEFAULT FALSE,
    
    -- Временные метки
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

-- Индексы для частых запросов
CREATE INDEX idx_mtr_item_type ON mtr_items(item_type);
CREATE INDEX idx_mtr_mtr_code ON mtr_items(mtr_code);
CREATE INDEX idx_mtr_ksm_code ON mtr_items(ksm_code);

-- GIN-индекс для JSONB-поля (позволяет искать по любым атрибутам)
CREATE INDEX idx_mtr_attributes ON mtr_items USING GIN(attributes);
```

Как это работает:

- Все специфичные поля (DN, PN, угол, толщина, материал, среда, покрытие) хранятся в attributes как {"dn": 150, "pn": 4.0, "angle": 90, "material": "09Г2С"}.
- При появлении нового поля (например, "coating_thickness") — просто добавляем его в JSONB, никаких изменений в схеме.
- Индекс GIN позволяет искать по любому ключу: attributes @> '{"dn": 150}'.

## Динамические справочники (для Этапа 1)

```sql
-- Ключевые слова для групп
CREATE TABLE group_keywords (
    id SERIAL PRIMARY KEY,
    group_name TEXT NOT NULL,
    keyword TEXT NOT NULL,
    priority INT DEFAULT 0,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT NOW()
);

-- Контекстные правила переопределения групп
CREATE TABLE contextual_overrides (
    id SERIAL PRIMARY KEY,
    trigger_phrase TEXT NOT NULL,
    target_group TEXT NOT NULL,
    priority INT DEFAULT 0,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT NOW()
);

-- Синонимы для нормализации
CREATE TABLE synonyms (
    id SERIAL PRIMARY KEY,
    group_name TEXT NOT NULL,          -- material, pressure_unit, climate, ...
    raw_value TEXT NOT NULL,
    normalized_value TEXT NOT NULL,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT NOW()
);

-- Константы для валидации (допустимые углы, PN для H2S и т.д.)
CREATE TABLE validation_constants (
    id SERIAL PRIMARY KEY,
    constant_name TEXT NOT NULL UNIQUE,
    value JSONB NOT NULL,
    description TEXT,
    updated_at TIMESTAMP DEFAULT NOW()
);

-- Правила валидации по типу детали
CREATE TABLE validation_rules (
    id SERIAL PRIMARY KEY,
    item_type TEXT NOT NULL UNIQUE,
    required_params TEXT[] NOT NULL,
    forbidden_params TEXT[] NOT NULL,
    optional_params TEXT[] NOT NULL,
    logical_conditions JSONB,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

-- Индексы для справочников
CREATE INDEX idx_group_keywords_group ON group_keywords(group_name);
CREATE INDEX idx_synonyms_group ON synonyms(group_name);
```

# 2. Neo4j — граф топологии

## Модель данных

Узлы:

```cypher
-- Компонент (деталь)
(:Component {
    ksm_code: "KSM-SYN-REG-000001",
    mtr_code: "MTR-SYN-REG-000001",
    item_type: "труба",
    dn: 500,
    pn: 16,
    material: "09ГСФ",
    medium: "газ с H2S и CO2",
    stock_qty: 19
})

-- Участок
(:Unit {
    unit_code: "UNIT-SYN-H2S-001",
    description: "Участок с сероводородом",
    medium: "H2S"
})

-- Компонент-идентификатор для COMP-SYN-XXX
(:ComponentAlias {
    comp_id: "COMP-SYN-010",
    ksm_code: "KSM-SYN-REG-000099",
    unit_code: "UNIT-SYN-H2S-001"
})
```

Рёбра:

```cypher
-- Соединение между компонентами
(:Component)-[:CONNECTS_TO {
    connection_type: "welded",   -- welded, flanged, threaded
    distance_m: 12.5,
    unit: "UNIT-SYN-H2S-001",
    template: "Gas compressor suction manifold",
    instance: 0,
    is_synthetic: true
}]->(:Component)

-- Принадлежность к участку
(:Component)-[:BELONGS_TO]->(:Unit)
```

Индексы и ограничения:

```cypher
CREATE CONSTRAINT FOR (c:Component) REQUIRE c.ksm_code IS UNIQUE;
CREATE CONSTRAINT FOR (u:Unit) REQUIRE u.unit_code IS UNIQUE;
CREATE CONSTRAINT FOR (a:ComponentAlias) REQUIRE a.comp_id IS UNIQUE;

CREATE INDEX FOR (c:Component) ON (c.item_type);
CREATE INDEX FOR (c:Component) ON (c.dn);
```
